# Ireland country-scoped runtime hooks. Startup initializes only IRE, while daily and monthly work stays on the exact scoped TAG on_action forms used by vanilla.

on_actions = {
	on_startup = {
		effect = {
			IRE = {
				if = {
					limit = {
						slopx_ireland_is_ireland = yes
						NOT = { has_country_flag = slopx_ireland_initialized }
					}
					slopx_ireland_initialize_campaign = yes
				}
			}
		}
	}

	on_daily_IRE = {

			}
		}
	}

	on_monthly_IRE = {

				limit = {
