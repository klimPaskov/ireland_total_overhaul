# Cabinet Briefing trigger contract.
# The four public cards are projections of the canonical Ireland ledgers. They
# do not create another balance layer, cache, selection state, or action gate.

ireland_mechanics_hub_initialized = {
	original_tag = IRE
	has_country_flag = ireland_initialized
}

ireland_mechanics_hub_open = {
	ireland_mechanics_hub_initialized = yes
	NOT = { has_country_flag = ireland_route_archived }
}

# Government is the public constitutional settlement and parliamentary
# programme card. It remains readable for the whole initialized Ireland route.
ireland_mechanics_hub_state_visible = {
	ireland_mechanics_hub_initialized = yes
}

ireland_mechanics_hub_state_critical = {
	ireland_mechanics_hub_state_visible = yes
	OR = {
		has_country_flag = ireland_constitutional_crisis_active
		has_country_flag = ireland_dual_authority_crisis_active
		check_variable = {
			var = ireland_national_settlement
			value = constant:ireland.settlement_stage_4_min
			compare = less_than
		}
	}
}

ireland_mechanics_hub_state_ready = {
	ireland_mechanics_hub_state_visible = yes
	NOT = { ireland_mechanics_hub_state_critical = yes }
	check_variable = {
		var = ireland_national_settlement
		value = constant:ireland.settlement_stage_5_min
		compare = greater_than_or_equals
	}
}

ireland_mechanics_hub_state_warning = {
	ireland_mechanics_hub_state_visible = yes
	NOT = { ireland_mechanics_hub_state_critical = yes }
	NOT = { ireland_mechanics_hub_state_ready = yes }
}

# National Supply is the existing Provision aggregate. Component shortages and
# bottleneck caps remain the authoritative reasons for a warning or critical
# state; the briefing never selects a component for the player.

	ireland_mechanics_hub_initialized = yes
	ireland_reveal_category_provision = yes
}

			var = ireland_provision_display

			compare = less_than
		}
	}
