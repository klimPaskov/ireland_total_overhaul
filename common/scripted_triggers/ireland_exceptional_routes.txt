	AND = {
		has_manpower > @IRELAND_EXCEPTIONAL_FIAN_MANPOWER_GATE
		has_equipment = { infantry_equipment > @IRELAND_EXCEPTIONAL_FIAN_INFANTRY_EQUIPMENT_GATE support_equipment > @IRELAND_EXCEPTIONAL_FIAN_SUPPORT_EQUIPMENT_GATE }
		command_power > @IRELAND_EXCEPTIONAL_COMMAND_LOW_GATE
		OR = {
			113 = {
				any_state_division = {
					OWNER = { tag = ROOT }
					NOT = { unit_strength < constant:ireland_exceptional_routes.fian_unit_strength_ready }
					NOT = { unit_organization < constant:ireland_exceptional_routes.fian_unit_organization_ready }
				}
			}
			134 = {
				any_state_division = {
					OWNER = { tag = ROOT }
					NOT = { unit_strength < constant:ireland_exceptional_routes.fian_unit_strength_ready }
					NOT = { unit_organization < constant:ireland_exceptional_routes.fian_unit_organization_ready }
