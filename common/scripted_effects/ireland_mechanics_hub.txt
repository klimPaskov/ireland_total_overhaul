# National Situation Ledger effects. These effects are presentation adapters:
# they recompute existing ledgers, select a bottleneck, or delegate to an
# existing guarded executor. They do not add a second balance or reward layer.

# Refresh the visible national report without advancing a route, review timer,
# core, hidden permission, political-power, or material ledger.
ireland_refresh_national_situation = {
	if = {
		limit = { ireland_mechanics_hub_open = yes }
		ireland_recompute_provision = yes
		ireland_recompute_readiness = yes
		ireland_sync_operational_state = yes
		if = {
			limit = { ireland_mechanics_hub_northern_visible = yes }
			ireland_sync_named_northern_objectives = yes
		}
		if = {
			limit = { ireland_mechanics_hub_culture_visible = yes }
			ireland_sync_cultural_evidence = yes
		}
		add_to_variable = { var = ireland_conditions_gui_dirty value = constant:ireland.one }
	}
}

# Execute exactly one next parliamentary programme step. The order matches the
# native Parliament panel and each canonical executor repeats its own guard.
ireland_execute_next_state_action = {
	if = {
		limit = { ireland_next_state_action_available = yes }
		if = {
			limit = { ireland_parliament_dail_action_available = yes }
			ireland_execute_parliament_dail_action = yes
		}
		else_if = {
			limit = { ireland_parliament_seanad_action_available = yes }
			ireland_execute_parliament_seanad_action = yes
		}
		else_if = {
			limit = { ireland_parliament_confidence_action_available = yes }
			ireland_execute_parliament_confidence_action = yes
		}
		else_if = {
			limit = { ireland_parliament_joint_budget_action_available = yes }
			ireland_execute_parliament_budget_action = yes
		}
		add_to_variable = { var = ireland_conditions_gui_dirty value = constant:ireland.one }
	}
}

# The canonical aggregate remains the existing rounded average with its
# bottleneck caps. This adapter only selects the first raw component minimum.
# Ties are deterministic in display order, so no cache value is exposed.
ireland_select_weakest_provision_component = {
	if = {
		limit = { ireland_mechanics_hub_provision_visible = yes }
		set_temp_variable = { var = ireland_mechanics_hub_weakest_value value = ireland_provision_food }
		set_variable = { var = ireland_mechanics_hub_selected_row value = constant:ireland_mechanics_hub.row_one }
		if = {
			limit = {
				check_variable = {
					var = ireland_provision_energy
					value = ireland_mechanics_hub_weakest_value
					compare = less_than
				}
			}
			set_temp_variable = { var = ireland_mechanics_hub_weakest_value value = ireland_provision_energy }
			set_variable = { var = ireland_mechanics_hub_selected_row value = constant:ireland_mechanics_hub.row_two }
		}
		if = {
			limit = {
				check_variable = {
					var = ireland_provision_transport
					value = ireland_mechanics_hub_weakest_value
					compare = less_than
				}
			}
			set_temp_variable = { var = ireland_mechanics_hub_weakest_value value = ireland_provision_transport }
			set_variable = { var = ireland_mechanics_hub_selected_row value = constant:ireland_mechanics_hub.row_three }
		}
		if = {
			limit = {
				check_variable = {
					var = ireland_provision_maritime
					value = ireland_mechanics_hub_weakest_value
					compare = less_than

		add_to_variable = { var = ireland_conditions_gui_dirty value = constant:ireland.one }
	}
}

# Northern and Evidence review adapters use the request flags already owned by
# the dedicated ledgers. Their existing synchroniser and final-review guards
# remain the only source of any review-time state change.
ireland_review_national_situation_northern = {
	if = {
		limit = { ireland_mechanics_hub_northern_review_available = yes }
		# Project the hub selection into the dedicated ledger before invoking its
		# canonical request consumer. This prevents a stale dedicated selection.
		set_variable = { var = ireland_northern_selected_row value = ireland_mechanics_hub_selected_row }
		set_country_flag = ireland_northern_review_requested
		ireland_sync_gui_requests = yes
		add_to_variable = { var = ireland_conditions_gui_dirty value = constant:ireland.one }
	}
}

ireland_review_national_situation_evidence = {
	if = {
		limit = { ireland_mechanics_hub_evidence_review_available = yes }
		# Evidence Record is a summary row. Preserve the player's dedicated-ledger
		# selection when one exists; otherwise inspect Provenance as the first
		# accountable evidence family instead of recording row zero.
		if = {
			limit = {
				NOT = {
					check_variable = {
						var = ireland_evidence_selected_row
						value = constant:ireland.zero
						compare = greater_than
					}
				}
			}
			set_variable = { var = ireland_evidence_selected_row value = constant:ireland.one }
		}
		set_country_flag = ireland_concealed_evidence_inspection_requested
		ireland_sync_gui_requests = yes
		add_to_variable = { var = ireland_conditions_gui_dirty value = constant:ireland.one }
	}
}

# Culture review before Evidence reveal recomputes only the public components.
# It selects the weakest public row and never calls hidden-permission
# synchronisation. The dispatcher routes a revealed Evidence row elsewhere.
ireland_review_national_situation_culture = {
	if = {
		limit = { ireland_mechanics_hub_culture_visible = yes }
		ireland_sync_cultural_evidence = yes
		ireland_select_weakest_culture_component = yes
		add_to_variable = { var = ireland_conditions_gui_dirty value = constant:ireland.one }
	}
}

# Dispatcher used by the full dashboard. Each branch is gated by the matching
# tab/action trigger, so a stale GUI click is a no-op and cannot bypass a cost.
ireland_execute_mechanics_hub_context_action = {
	if = {
		limit = { ireland_mechanics_hub_context_action_available = yes }
		if = {
			limit = { check_variable = { var = ireland_mechanics_hub_selected_tab value = constant:ireland_mechanics_hub.tab_state compare = equals } }
			ireland_execute_next_state_action = yes
		}
		else_if = {
			limit = { check_variable = { var = ireland_mechanics_hub_selected_tab value = constant:ireland_mechanics_hub.tab_provision compare = equals } }
			ireland_recompute_provision = yes
			ireland_select_weakest_provision_component = yes
		}
		else_if = {
			limit = { check_variable = { var = ireland_mechanics_hub_selected_tab value = constant:ireland_mechanics_hub.tab_readiness compare = equals } }
			ireland_recompute_readiness = yes
			ireland_select_weakest_readiness_component = yes
		}
		else_if = {
			limit = { check_variable = { var = ireland_mechanics_hub_selected_tab value = constant:ireland_mechanics_hub.tab_foreign compare = equals } }
			ireland_review_national_situation_foreign = yes
		}
		else_if = {
			limit = { check_variable = { var = ireland_mechanics_hub_selected_tab value = constant:ireland_mechanics_hub.tab_northern compare = equals } }
			ireland_review_national_situation_northern = yes
		}
		else_if = {
			limit = { check_variable = { var = ireland_mechanics_hub_selected_tab value = constant:ireland_mechanics_hub.tab_culture compare = equals } }
			if = {
				limit = { ireland_mechanics_hub_evidence_review_available = yes }
				ireland_review_national_situation_evidence = yes
			}
			else = { ireland_review_national_situation_culture = yes }
		}
		add_to_variable = { var = ireland_conditions_gui_dirty value = constant:ireland.one }
	}
}
