# Cabinet Briefing effects.
# The briefing is presentation-only. Its sole action delegates to the existing
# guarded Parliament executors and never mutates a second ledger.

# Execute exactly one next parliamentary programme step. The order matches the
# native Parliament panel; each canonical executor repeats its own guard so a
# stale GUI click cannot bypass a requirement or cost.
ireland_execute_next_state_action = {
	if = {
		limit = { ireland_briefing_context_action_available = yes }
		if = {
			limit = { ireland_parliament_dail_action_available = yes }
			ireland_execute_parliament_dail_action = yes
		}
		else_if = {
			limit = { ireland_parliament_seanad_action_available = yes }
			ireland_execute_parliament_seanad_action = yes
		}
		else_if = {
			limit = { ireland_parliament_confidence_action_available = yes }
			ireland_execute_parliament_confidence_action = yes
		}
		else_if = {
			limit = { ireland_parliament_joint_budget_action_available = yes }
			ireland_execute_parliament_budget_action = yes
		}
		add_to_variable = { var = ireland_conditions_gui_dirty value = constant:ireland.one }
	}
}
