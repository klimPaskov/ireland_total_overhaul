# Ireland operational-state synchronisation. This file turns canonical ledgers and
# incident source flags into the active signals consumed by laws, missions, offices,
# decisions, and the three Ireland scripted GUIs. It contains no passive rewards.

ireland_sync_operational_state = {
	# A war or a zeroed provision/readiness component is an active emergency. The
	# dated emergency route remains active until its repeal review, while a real
	# shortage can reopen the state after repeal and must be recovered in play.
	if = {
		limit = {
			OR = {
				has_war = yes
				check_variable = { var = ireland_provision_food value = constant:ireland.zero compare = equals }
				check_variable = { var = ireland_provision_energy value = constant:ireland.zero compare = equals }
				check_variable = { var = ireland_provision_transport value = constant:ireland.zero compare = equals }
				check_variable = { var = ireland_provision_maritime value = constant:ireland.zero compare = equals }
				check_variable = { var = ireland_readiness_command value = constant:ireland.zero compare = equals }
				check_variable = { var = ireland_readiness_equipment value = constant:ireland.zero compare = equals }
				check_variable = { var = ireland_readiness_mobilisation value = constant:ireland.zero compare = equals }
				check_variable = { var = ireland_readiness_supply value = constant:ireland.zero compare = equals }
				check_variable = { var = ireland_readiness_warning value = constant:ireland.zero compare = equals }
				check_variable = { var = ireland_readiness_maritime_protection value = constant:ireland.zero compare = equals }
				check_variable = { var = ireland_readiness_intelligence value = constant:ireland.zero compare = equals }
				AND = {
					NOT = { has_country_flag = ireland_emergency_repeal_complete }
					OR = {
						has_country_flag = ireland_emergency_opening
						has_country_flag = ireland_statutory_emergency_complete
					}
				}
			}
		}
		set_country_flag = ireland_national_emergency_active
	}
	if = {
		limit = {
			has_country_flag = ireland_national_emergency_active
			NOT = { has_war = yes }
			NOT = { has_country_flag = ireland_emergency_opening }
			NOT = { has_country_flag = ireland_statutory_emergency_complete }
			has_country_flag = ireland_emergency_repeal_complete
			NOT = {
				OR = {
					check_variable = { var = ireland_provision_food value = constant:ireland.zero compare = equals }
					check_variable = { var = ireland_provision_energy value = constant:ireland.zero compare = equals }
					check_variable = { var = ireland_provision_transport value = constant:ireland.zero compare = equals }
					check_variable = { var = ireland_provision_maritime value = constant:ireland.zero compare = equals }
					check_variable = { var = ireland_readiness_command value = constant:ireland.zero compare = equals }
					check_variable = { var = ireland_readiness_equipment value = constant:ireland.zero compare = equals }
					check_variable = { var = ireland_readiness_mobilisation value = constant:ireland.zero compare = equals }
					check_variable = { var = ireland_readiness_supply value = constant:ireland.zero compare = equals }
					check_variable = { var = ireland_readiness_warning value = constant:ireland.zero compare = equals }
					check_variable = { var = ireland_readiness_maritime_protection value = constant:ireland.zero compare = equals }
					check_variable = { var = ireland_readiness_intelligence value = constant:ireland.zero compare = equals }
				}
			}
		}
		clr_country_flag = ireland_national_emergency_active
	}
	if = {
		limit = {
			OR = {
				has_country_flag = ireland_national_emergency_active
				has_war = yes
			}
		}
		set_country_flag = ireland_external_emergency
	}
	else = {
		clr_country_flag = ireland_external_emergency
	}

	# Command and mobilisation failures are active conditions, not rewards. The
	# durable achievement history helper records them before recovery clears the
	# operational vacancy signal.
	if = {
		limit = { check_variable = { var = ireland_readiness_command value = constant:ireland.zero compare = equals } }
		set_country_flag = ireland_military_command_collapse
	}
	else = {
		clr_country_flag = ireland_military_command_collapse
	}
	if = {
		limit = { check_variable = { var = ireland_security_loyalty value = constant:ireland.zero compare = equals } }
		set_country_flag = ireland_security_command_collapse
	}
	else = {
		clr_country_flag = ireland_security_command_collapse
	}
	if = {
		limit = { check_variable = { var = ireland_readiness_mobilisation value = constant:ireland.zero compare = equals } }
		set_country_flag = ireland_mobilisation_failure
		set_country_flag = ireland_mobilisation_command_collapse
	}
	else = {
		clr_country_flag = ireland_mobilisation_failure
		clr_country_flag = ireland_mobilisation_command_collapse
	}

	ireland_sync_neutrality_incident_family = yes
	add_to_variable = { var = ireland_conditions_gui_dirty value = constant:ireland.one }
}
