ireland_finalize_route_conclusion = {
	# This helper evaluates the due review and dispatches exactly one family
	# report. It is intentionally non-terminal: only a successful report option
	# may call ireland_archive_successful_route.
	if = {
		limit = {
			has_country_flag = ireland_conclusion_review_due
			NOT = { has_country_flag = ireland_route_archived }
			ireland_has_route_conclusion_capstone = yes
		}
		clr_country_flag = ireland_route_conclusion_full
		clr_country_flag = ireland_route_conclusion_partial
		clr_country_flag = ireland_route_conclusion_blocked
		clr_country_flag = ireland_route_conclusion_failed
		clr_country_flag = ireland_route_conclusion_blocked_government
		clr_country_flag = ireland_route_conclusion_blocked_law
		clr_country_flag = ireland_route_conclusion_blocked_economy
		clr_country_flag = ireland_route_conclusion_blocked_defence
		clr_country_flag = ireland_route_conclusion_blocked_foreign
		clr_country_flag = ireland_route_conclusion_blocked_northern
		clr_country_flag = ireland_route_conclusion_blocked_identity
		clr_country_flag = ireland_route_conclusion_blocked_objectives
		set_variable = { var = ireland_route_conclusion_result value = constant:ireland.route_conclusion_result_none }
		set_country_flag = ireland_route_conclusion_eligibility_archived

		# A live collapse or an unrecovered route failure is a failed conclusion,
		# not a cosmetic victory. The history flags remain available to recovery.
		if = {
			limit = {
				OR = {
					has_country_flag = ireland_emergency_overreach
					has_country_flag = ireland_military_command_collapse
					has_country_flag = ireland_security_command_collapse
					has_country_flag = ireland_northern_settlement_failure
					AND = {
						has_country_flag = ireland_route_failure
						NOT = { has_country_flag = ireland_route_failure_recovered }
					}
				}
			}
			set_variable = { var = ireland_route_conclusion_result value = constant:ireland.route_conclusion_result_failed }
			set_country_flag = ireland_route_conclusion_failed
			set_country_flag = ireland_route_failure
		}
		else = {
			# Each flag is a named blocked surface for the postwar report. This
			# makes a blocked review inspectable without exposing implementation
			# variables in ordinary focus or event prose.
			if = {
				limit = {
					NOT = { ireland_family_postwar_government_repair_recorded = yes }
					OR = {
						NOT = { has_country_flag = ireland_route_leader_transition_complete }
						has_country_flag = ireland_office_vacancy
						check_variable = { var = ireland_office_vacancy_count value = constant:ireland.zero compare = greater_than }
					}
				}
				set_country_flag = ireland_route_conclusion_blocked_government
			}
			if = {
				limit = {
					NOT = { ireland_family_postwar_law_repair_recorded = yes }
					NOT = { has_country_flag = ireland_conclusion_proof }
				}
				set_country_flag = ireland_route_conclusion_blocked_law
			}
			if = {
				limit = {
					NOT = { ireland_family_postwar_economy_repair_recorded = yes }
					NOT = { has_country_flag = ireland_provision_conclusion_ready }
				}
				set_country_flag = ireland_route_conclusion_blocked_economy
			}
			if = {
				limit = {
					NOT = { ireland_family_postwar_defence_repair_recorded = yes }
					NOT = { has_country_flag = ireland_readiness_conclusion_ready }
				}
				set_country_flag = ireland_route_conclusion_blocked_defence
			}
			if = {
				limit = {
					OR = {
						NOT = { has_country_flag = ireland_foreign_posture_selected }
						has_country_flag = ireland_foreign_reaction_window_active
						has_country_flag = ireland_neutrality_incident_active
					}
				}
				set_country_flag = ireland_route_conclusion_blocked_foreign
			}
			if = {
				limit = {
					NOT = {
						OR = {
							has_country_flag = ireland_northern_policy_selected
							has_country_flag = ireland_northern_deferment_accepted
						}
					}
				}
				set_country_flag = ireland_route_conclusion_blocked_northern
			}
			if = {
				limit = {
					has_country_flag = ireland_route_identity_sync_required
					NOT = { ireland_family_postwar_identity_repair_recorded = yes }
				}
				set_country_flag = ireland_route_conclusion_blocked_identity
			}
			if = {
				limit = {
					OR = {
						check_variable = { var = ireland_active_mission_count value = constant:ireland.zero compare = greater_than }
						NOT = { ireland_family_postwar_settlement_complete = yes }
					}
				}
				set_country_flag = ireland_route_conclusion_blocked_objectives
			}
			if = {
				limit = {
					OR = {
						has_country_flag = ireland_route_conclusion_blocked_government
						has_country_flag = ireland_route_conclusion_blocked_law
						has_country_flag = ireland_route_conclusion_blocked_economy
						has_country_flag = ireland_route_conclusion_blocked_defence
						has_country_flag = ireland_route_conclusion_blocked_foreign
						has_country_flag = ireland_route_conclusion_blocked_northern
						has_country_flag = ireland_route_conclusion_blocked_identity
						has_country_flag = ireland_route_conclusion_blocked_objectives
					}
				}
				set_variable = { var = ireland_route_conclusion_result value = constant:ireland.route_conclusion_result_blocked }
				set_country_flag = ireland_route_conclusion_blocked
				set_country_flag = ireland_route_failure
			}
			else = {
				# Partial success remembers damaged settlement, prior recovery, and
				# partial mission outcomes. It never upgrades a partial record later.
				if = {
					limit = {
						OR = {
							check_variable = { var = ireland_missions_partial value = constant:ireland.zero compare = greater_than }
							has_country_flag = ireland_route_failure_recovered
							check_variable = { var = ireland_national_settlement value = constant:ireland.settlement_stage_5_min compare = less_than }
							check_variable = { var = ireland_social_consent value = constant:ireland.vocational_mandate_consent_min compare = less_than }
						}
					}
					set_variable = { var = ireland_route_conclusion_result value = constant:ireland.route_conclusion_result_partial }
					set_country_flag = ireland_route_conclusion_partial
				}
				else = {
					set_variable = { var = ireland_route_conclusion_result value = constant:ireland.route_conclusion_result_full }
					set_country_flag = ireland_route_conclusion_full
				}
			}
		}

		ireland_record_route_conclusion_flags = yes

		# The Clann agrarian broker concludes through the host constitutional
		# order. It receives a durable qualification only when the shared review
		# records full or partial material success; failed and blocked reviews do
		# not become political victories.
		if = {
			limit = {
				has_country_flag = ireland_route_clann_agrarian_broker
				has_country_flag = ireland_clann_host_conclusion_ready
				OR = {
					has_country_flag = ireland_route_conclusion_full
					has_country_flag = ireland_route_conclusion_partial
				}
			}
			set_country_flag = ireland_clann_agrarian_broker_concluded
			set_country_flag = ireland_clann_agrarian_programme_government_qualified
		}

		clr_country_flag = ireland_conclusion_review_active
		clr_country_flag = ireland_conclusion_review_due
		set_variable = { var = ireland_conclusion_review_days value = constant:ireland.zero }
		set_country_flag = ireland_route_conclusion_review_complete
		if = {
			limit = {
				OR = {
					has_country_flag = ireland_route_conclusion_blocked
					has_country_flag = ireland_route_conclusion_failed
				}
			}
			ireland_open_family_conclusion_recovery = yes
		}
		ireland_dispatch_family_conclusion_event = yes
	}
}

ireland_update_mission_capacity = {
	set_temp_variable = { var = ireland_previous_mission_capacity value = ireland_mission_capacity }
	set_variable = { var = ireland_mission_capacity value = constant:ireland.mission_cap_normal }
	if = {
		limit = {
			OR = {
				has_country_flag = ireland_national_emergency_active
				has_war = yes
			}
		}
		set_variable = { var = ireland_mission_capacity value = constant:ireland.mission_cap_emergency_or_war }
	}
	if = {
		limit = {
			has_country_flag = ireland_national_emergency_active
			has_country_flag = ireland_northern_occupation_active
		}
		set_variable = { var = ireland_mission_capacity value = constant:ireland.mission_cap_emergency_occupation }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = ireland_national_emergency_active
				has_war = yes
			}
		}
		set_variable = { var = ireland_mission_stable_days value = constant:ireland.zero }
	}
	else = {
		if = {
			limit = {
				check_variable = {
					var = ireland_previous_mission_capacity
					value = constant:ireland.mission_cap_emergency_or_war
					compare = greater_than_or_equals
				}
			}
			# Leaving either emergency band first falls back to the 10-slot
			# band. The normal eight-slot cap returns only after thirty
			# consecutive stable days, including when the previous state was 12.
			set_variable = { var = ireland_mission_capacity value = constant:ireland.mission_cap_emergency_or_war }
			if = {
				limit = {
					check_variable = {
						var = ireland_mission_stable_days
						value = constant:ireland.mission_capacity_stable_days
						compare = less_than
					}
				}
				add_to_variable = { var = ireland_mission_stable_days value = constant:ireland.one }
			}
			if = {
				limit = {
					check_variable = {
						var = ireland_mission_stable_days
						value = constant:ireland.mission_capacity_stable_days
						compare = greater_than_or_equals
					}
				}
				set_variable = { var = ireland_mission_capacity value = constant:ireland.mission_cap_normal }
			}
		}
		else = {
			set_variable = { var = ireland_mission_stable_days value = constant:ireland.zero }
		}
	}
	ireland_rebuild_mission_counters = yes
	ireland_admit_critical_mission = yes
}
