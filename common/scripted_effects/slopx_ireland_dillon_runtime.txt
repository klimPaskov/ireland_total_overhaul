# Dillon route boundary helpers. These remain separate from the broad runtime
# registry so the native National Settlement balance-of-power guard is easy to
# audit beside the route-specific records it protects.

slopx_ireland_sync_dillon_crisis_boundary = {
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_route_cleanup
				NOT = { has_country_flag = slopx_ireland_route_dillon_intervention }
			}
		}
		clr_country_flag = slopx_ireland_dillon_vote_passed
	}
	if = {
		limit = {
			is_power_balance_in_range = {
				id = slopx_ireland_national_settlement
				range = slopx_ireland_armed_dual_authority_range
			}
		}
		set_country_flag = slopx_ireland_dual_authority_crisis_active
		set_country_flag = slopx_ireland_constitutional_crisis_active
	}
	else = {
		clr_country_flag = slopx_ireland_dual_authority_crisis_active
	}
}
