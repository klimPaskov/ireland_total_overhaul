# Public culture and concealed-evidence ledgers for the Ireland package.
# The public components are recomputed from the living Cultural Capacity ledger
# and named institutional work. Evidence counters are durable projections of
# independent records, sites, regions, witnesses, and obligations; no counter
# is advanced by merely selecting a hidden focus.

slopx_ireland_sync_cultural_evidence = {
	# Keep the four visible components distinct even while Cultural Capacity is
	# their presentation summary. Institutional work closes the opening gaps and
	# missions provide the larger step changes.
	set_variable = { var = slopx_ireland_cultural_language_service value = slopx_ireland_cultural_capacity }
	set_variable = { var = slopx_ireland_cultural_archival_record value = slopx_ireland_cultural_capacity }
	subtract_from_variable = { slopx_ireland_cultural_archival_record = constant:slopx_ireland.cultural_component_archival_offset }
	set_variable = { var = slopx_ireland_cultural_regional_participation value = slopx_ireland_cultural_capacity }
	subtract_from_variable = { slopx_ireland_cultural_regional_participation = constant:slopx_ireland.cultural_component_regional_offset }
	set_variable = { var = slopx_ireland_cultural_site_stewardship value = slopx_ireland_cultural_capacity }
	subtract_from_variable = { slopx_ireland_cultural_site_stewardship = constant:slopx_ireland.cultural_component_site_offset }

	if = {
		limit = { has_country_flag = slopx_ireland_language_service_complete }
		add_to_variable = { slopx_ireland_cultural_language_service = constant:slopx_ireland.cultural_focus_bonus }
	}
	if = {
		limit = { has_country_flag = slopx_ireland_language_institutions_complete }
		add_to_variable = { slopx_ireland_cultural_language_service = constant:slopx_ireland.cultural_focus_bonus }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_action_fund_gaeltacht_administration_complete
				has_country_flag = slopx_ireland_action_standardise_irish_language_services_complete
			}
		}
		add_to_variable = { slopx_ireland_cultural_language_service = constant:slopx_ireland.cultural_action_bonus }
	}

	if = {
		limit = { has_country_flag = slopx_ireland_civil_war_memory_complete }
		add_to_variable = { slopx_ireland_cultural_archival_record = constant:slopx_ireland.cultural_focus_bonus }
	}
	if = {
		limit = { has_country_flag = slopx_ireland_archival_record_complete }
		add_to_variable = { slopx_ireland_cultural_archival_record = constant:slopx_ireland.cultural_focus_bonus }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_action_survey_provincial_archives_complete
				has_country_flag = slopx_ireland_action_expand_folklore_collection_complete
			}
		}
		add_to_variable = { slopx_ireland_cultural_archival_record = constant:slopx_ireland.cultural_action_bonus }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_mission_collect_a_national_folklore_cycle_full_recorded
				has_country_flag = slopx_ireland_mission_verify_the_restoration_evidence_full_recorded
			}
		}
		add_to_variable = { slopx_ireland_cultural_archival_record = constant:slopx_ireland.cultural_mission_bonus }
	}

	if = {
		limit = { has_country_flag = slopx_ireland_border_observation_complete }
		add_to_variable = { slopx_ireland_cultural_regional_participation = constant:slopx_ireland.cultural_focus_bonus }
	}
	if = {
		limit = { has_country_flag = slopx_ireland_regional_participation_complete }
		add_to_variable = { slopx_ireland_cultural_regional_participation = constant:slopx_ireland.cultural_focus_bonus }
	}
	if = {
		limit = { has_country_flag = slopx_ireland_diaspora_exchange_complete }
		add_to_variable = { slopx_ireland_cultural_regional_participation = constant:slopx_ireland.cultural_focus_bonus }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_action_hold_a_provincial_cultural_congress_complete
				has_country_flag = slopx_ireland_action_invite_northern_delegations_complete
			}
		}
		add_to_variable = { slopx_ireland_cultural_regional_participation = constant:slopx_ireland.cultural_action_bonus }
	}
	if = {
		limit = { has_country_flag = slopx_ireland_mission_hold_a_five_region_cultural_congress_full_recorded }
		add_to_variable = { slopx_ireland_cultural_regional_participation = constant:slopx_ireland.cultural_mission_bonus }
	}

	if = {
		limit = { has_country_flag = slopx_ireland_site_stewardship_complete }
		add_to_variable = { slopx_ireland_cultural_site_stewardship = constant:slopx_ireland.cultural_focus_bonus }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_action_protect_a_historic_site_complete
				has_country_flag = slopx_ireland_mission_fulfil_the_compact_at_five_sites_full_recorded
			}
		}
		add_to_variable = { slopx_ireland_cultural_site_stewardship = constant:slopx_ireland.cultural_action_bonus }
	}

	clamp_variable = { var = slopx_ireland_cultural_language_service min = constant:slopx_ireland.value_min max = constant:slopx_ireland.value_max }
	clamp_variable = { var = slopx_ireland_cultural_archival_record min = constant:slopx_ireland.value_min max = constant:slopx_ireland.value_max }
	clamp_variable = { var = slopx_ireland_cultural_regional_participation min = constant:slopx_ireland.value_min max = constant:slopx_ireland.value_max }
	clamp_variable = { var = slopx_ireland_cultural_site_stewardship min = constant:slopx_ireland.value_min max = constant:slopx_ireland.value_max }
	clamp_variable = { var = slopx_ireland_cultural_capacity min = constant:slopx_ireland.value_min max = constant:slopx_ireland.value_max }

	# Public lane availability is derived from completed H/P institutions, not
	# from a hidden route flag. This makes the culture category playable before
	# the concealed branch is ever considered.
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_language_service_complete
				has_country_flag = slopx_ireland_cultural_survey_complete
			}
		}
		set_country_flag = slopx_ireland_serious_culture_active
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_civil_war_memory_complete
				has_country_flag = slopx_ireland_cultural_survey_complete
			}
		}
		set_country_flag = slopx_ireland_archive_lane_open
	}
	if = {
		limit = {
			has_country_flag = slopx_ireland_cultural_survey_complete
			has_country_flag = slopx_ireland_language_service_complete
		}
		set_country_flag = slopx_ireland_language_capacity_ready
	}
	if = {
		limit = {
			has_country_flag = slopx_ireland_cultural_survey_complete
			has_country_flag = slopx_ireland_civil_war_memory_complete
		}
		set_country_flag = slopx_ireland_archival_capacity_ready
	}
	if = {
		limit = {
			has_country_flag = slopx_ireland_cultural_survey_complete
			has_country_flag = slopx_ireland_border_observation_complete
		}
		set_country_flag = slopx_ireland_regional_institutions_ready
	}
	if = {
		limit = {
			has_country_flag = slopx_ireland_archival_record_complete
			has_country_flag = slopx_ireland_regional_participation_complete
		}
		set_country_flag = slopx_ireland_site_stewardship_ready
	}
	if = {
		limit = {
			has_country_flag = slopx_ireland_site_stewardship_complete
			has_country_flag = slopx_ireland_northern_contacts_complete
		}
		set_country_flag = slopx_ireland_diaspora_network_ready
	}

	# Count independent evidence classes from named work. These are deliberately
	# projections, so a failed or partial mission never becomes a full record.
	set_variable = { var = slopx_ireland_concealed_evidence_credible_count value = constant:slopx_ireland.zero }
	if = { limit = { has_country_flag = slopx_ireland_language_service_complete } add_to_variable = { slopx_ireland_concealed_evidence_credible_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_archival_record_complete } add_to_variable = { slopx_ireland_concealed_evidence_credible_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_regional_participation_complete } add_to_variable = { slopx_ireland_concealed_evidence_credible_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_site_stewardship_complete } add_to_variable = { slopx_ireland_concealed_evidence_credible_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_diaspora_exchange_complete } add_to_variable = { slopx_ireland_concealed_evidence_credible_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_action_survey_provincial_archives_complete } add_to_variable = { slopx_ireland_concealed_evidence_credible_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_mission_collect_a_national_folklore_cycle_full_recorded } add_to_variable = { slopx_ireland_concealed_evidence_credible_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_mission_verify_the_restoration_evidence_full_recorded } add_to_variable = { slopx_ireland_concealed_evidence_credible_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_otherworld_witnesses_complete } add_to_variable = { slopx_ireland_concealed_evidence_credible_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_cultural_evidence_ledger_complete } add_to_variable = { slopx_ireland_concealed_evidence_credible_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_otherworld_evidence_ledger_complete } add_to_variable = { slopx_ireland_concealed_evidence_credible_count = constant:slopx_ireland.one } }
	clamp_variable = { var = slopx_ireland_concealed_evidence_credible_count min = constant:slopx_ireland.value_min max = constant:slopx_ireland.cultural_evidence_max }

	set_variable = { var = slopx_ireland_concealed_evidence_category_count value = constant:slopx_ireland.zero }
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_archival_record_complete
				has_country_flag = slopx_ireland_action_survey_provincial_archives_complete
				has_country_flag = slopx_ireland_mission_collect_a_national_folklore_cycle_full_recorded
			}
		}
		add_to_variable = { slopx_ireland_concealed_evidence_category_count = constant:slopx_ireland.one }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_constitutional_draft_complete
				has_country_flag = slopx_ireland_dail_supremacy_clause_complete
				has_country_flag = slopx_ireland_high_king_convention_complete
				has_country_flag = slopx_ireland_cultural_evidence_ledger_complete
			}
		}
		add_to_variable = { slopx_ireland_concealed_evidence_category_count = constant:slopx_ireland.one }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_regional_participation_complete
				has_country_flag = slopx_ireland_action_hold_a_provincial_cultural_congress_complete
				has_country_flag = slopx_ireland_mission_hold_a_five_region_cultural_congress_full_recorded
			}
		}
		add_to_variable = { slopx_ireland_concealed_evidence_category_count = constant:slopx_ireland.one }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_site_stewardship_complete
				has_country_flag = slopx_ireland_action_protect_a_historic_site_complete
				has_country_flag = slopx_ireland_mission_fulfil_the_compact_at_five_sites_full_recorded
			}
		}
		add_to_variable = { slopx_ireland_concealed_evidence_category_count = constant:slopx_ireland.one }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_otherworld_witnesses_complete
				has_country_flag = slopx_ireland_independent_witnesses_ready
				has_country_flag = slopx_ireland_diaspora_exchange_complete
			}
		}
		add_to_variable = { slopx_ireland_concealed_evidence_category_count = constant:slopx_ireland.one }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_action_test_a_restoration_claim_complete
				has_country_flag = slopx_ireland_high_king_evidence_gate_complete
			}
		}
		add_to_variable = { slopx_ireland_concealed_evidence_category_count = constant:slopx_ireland.one }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_fivefold_obligation_complete
				has_country_flag = slopx_ireland_compact_with_otherworld_complete
				has_country_flag = slopx_ireland_otherworld_evidence_ledger_complete
			}
		}
		add_to_variable = { slopx_ireland_concealed_evidence_category_count = constant:slopx_ireland.one }
	}
	clamp_variable = { var = slopx_ireland_concealed_evidence_category_count min = constant:slopx_ireland.value_min max = constant:slopx_ireland.cultural_evidence_category_max }

	set_variable = { var = slopx_ireland_concealed_evidence_region_count value = constant:slopx_ireland.zero }
	if = { limit = { has_country_flag = slopx_ireland_regional_participation_complete } add_to_variable = { slopx_ireland_concealed_evidence_region_count = constant:slopx_ireland.cultural_region_participation_regions } }
	if = { limit = { has_country_flag = slopx_ireland_northern_contacts_complete } add_to_variable = { slopx_ireland_concealed_evidence_region_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_mission_collect_a_national_folklore_cycle_full_recorded } add_to_variable = { slopx_ireland_concealed_evidence_region_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_mission_hold_a_five_region_cultural_congress_full_recorded } add_to_variable = { slopx_ireland_concealed_evidence_region_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_action_protect_a_historic_site_complete } add_to_variable = { slopx_ireland_concealed_evidence_region_count = constant:slopx_ireland.one } }
	if = { limit = { has_country_flag = slopx_ireland_mission_verify_the_restoration_evidence_full_recorded } add_to_variable = { slopx_ireland_concealed_evidence_region_count = constant:slopx_ireland.one } }
	clamp_variable = { var = slopx_ireland_concealed_evidence_region_count min = constant:slopx_ireland.value_min max = constant:slopx_ireland.cultural_evidence_region_max }

	set_variable = { var = slopx_ireland_concealed_evidence_disqualifier_count value = constant:slopx_ireland.zero }
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_achievement_fabricated_evidence
				has_country_flag = slopx_ireland_achievement_hoax
				has_country_flag = slopx_ireland_achievement_government_fabrication
				has_country_flag = slopx_ireland_achievement_foreign_control
				has_country_flag = slopx_ireland_achievement_witness_persecution
				has_country_flag = slopx_ireland_religious_persecution_failure
				has_country_flag = slopx_ireland_achievement_campaign_disqualified
			}
		}
		add_to_variable = { slopx_ireland_concealed_evidence_disqualifier_count = constant:slopx_ireland.one }
	}
	clamp_variable = { var = slopx_ireland_concealed_evidence_disqualifier_count min = constant:slopx_ireland.value_min max = constant:slopx_ireland.value_max }

	set_variable = { var = slopx_ireland_concealed_evidence_obligation_count value = slopx_ireland_achievement_accepted_obligation_count }
	if = {
		limit = { has_country_flag = slopx_ireland_fivefold_obligation_complete NOT = { has_country_flag = slopx_ireland_concealed_fivefold_obligation_recorded } }
		add_to_variable = { slopx_ireland_concealed_evidence_obligation_count = constant:slopx_ireland.one }
		set_country_flag = slopx_ireland_concealed_fivefold_obligation_recorded
	}
	if = {
		limit = { has_country_flag = slopx_ireland_compact_with_otherworld_complete NOT = { has_country_flag = slopx_ireland_concealed_otherworld_obligation_recorded } }
		add_to_variable = { slopx_ireland_concealed_evidence_obligation_count = constant:slopx_ireland.one }
		set_country_flag = slopx_ireland_concealed_otherworld_obligation_recorded
	}
	clamp_variable = { var = slopx_ireland_concealed_evidence_obligation_count min = constant:slopx_ireland.value_min max = constant:slopx_ireland.cultural_evidence_max }

	set_variable = { var = slopx_ireland_concealed_evidence_public_knowledge value = slopx_ireland_public_knowledge }
	clamp_variable = { var = slopx_ireland_public_knowledge min = constant:slopx_ireland.value_min max = constant:slopx_ireland.cultural_public_knowledge_max }
	clamp_variable = { var = slopx_ireland_concealed_evidence_public_knowledge min = constant:slopx_ireland.value_min max = constant:slopx_ireland.cultural_public_knowledge_max }

	# The first credible item opens the ledger. Route verification remains gated
	# by the separate counters above and by the route-specific helper in runtime.
	if = {
		limit = {
			has_country_flag = slopx_ireland_cultural_evidence_ledger_complete
			check_variable = { var = slopx_ireland_concealed_evidence_credible_count value = constant:slopx_ireland.zero compare = greater_than }
		}
		set_country_flag = slopx_ireland_hidden_evidence_gate
	}
	if = {
		limit = {
			has_country_flag = slopx_ireland_cultural_service_complete
			has_country_flag = slopx_ireland_diaspora_exchange_complete
			check_variable = { var = slopx_ireland_concealed_evidence_credible_count value = constant:slopx_ireland.cultural_evidence_royal_credible compare = greater_than_or_equals }
			check_variable = { var = slopx_ireland_concealed_evidence_category_count value = constant:slopx_ireland.cultural_evidence_royal_categories compare = greater_than_or_equals }
			check_variable = { var = slopx_ireland_concealed_evidence_region_count value = constant:slopx_ireland.cultural_evidence_royal_regions compare = greater_than_or_equals }
			OR = {
				has_country_flag = slopx_ireland_mission_verify_the_restoration_evidence_full_recorded
				has_country_flag = slopx_ireland_achievement_independent_evidence_provenance
			}
		}
		set_country_flag = slopx_ireland_independent_evidence_ready
	}
	add_to_variable = { var = slopx_ireland_evidence_gui_dirty value = constant:slopx_ireland.one }
}

# Commit the explicit Otherworld archive reveal. The event owns the player
# choice; this helper owns the route-specific H/P-to-A boundary and the two subsequent
# institutional lanes so no option can omit a required durable flag.
	slopx_ireland_commit_otherworld_archive_reveal = {
	set_country_flag = slopx_ireland_otherworld_archive_reveal_complete
	set_country_flag = slopx_ireland_otherworld_impossible_confirmed
	set_country_flag = slopx_ireland_otherworld_compact_ready
	set_country_flag = slopx_ireland_otherworld_containment_ready
}

slopx_ireland_sync_plebiscite_state = {
	if = {
		limit = {
			has_country_flag = slopx_ireland_plebiscite_campaign_open
			NOT = {
				OR = {
					has_country_flag = slopx_ireland_plebiscite_passed
					has_country_flag = slopx_ireland_plebiscite_rejected
					has_country_flag = slopx_ireland_plebiscite_recovered
				}
			}
		}
		set_country_flag = slopx_ireland_plebiscite_open
		set_country_flag = slopx_ireland_northern_plebiscite_open
	}
	if = {
		limit = {
			OR = {
				has_country_flag = slopx_ireland_plebiscite_passed
				has_country_flag = slopx_ireland_plebiscite_rejected
				has_country_flag = slopx_ireland_plebiscite_recovered
			}
		}
		clr_country_flag = slopx_ireland_plebiscite_open
		clr_country_flag = slopx_ireland_northern_plebiscite_open
		clr_country_flag = slopx_ireland_plebiscite_campaign_open
	}
}

	if = {
		limit = { has_country_flag = slopx_ireland_conditions_review_requested }

		slopx_ireland_recompute_provision = yes
		slopx_ireland_recompute_readiness = yes
		slopx_ireland_sync_operational_state = yes
		slopx_ireland_sync_law_registry = yes
		clr_country_flag = slopx_ireland_conditions_review_requested
		add_to_variable = { slopx_ireland_conditions_gui_dirty = constant:slopx_ireland.one }
	}
	if = {
		limit = { has_country_flag = slopx_ireland_northern_review_requested }
		set_variable = { var = slopx_ireland_northern_last_reviewed_row value = slopx_ireland_northern_selected_row }
		slopx_ireland_sync_northern_ledger = yes
		slopx_ireland_apply_northern_core_review = yes
		slopx_ireland_update_achievement_trackers = yes
		clr_country_flag = slopx_ireland_northern_review_requested
		add_to_variable = { slopx_ireland_northern_gui_dirty = constant:slopx_ireland.one }
	}
	if = {
		limit = { has_country_flag = slopx_ireland_concealed_evidence_inspection_requested }
		set_variable = { var = slopx_ireland_evidence_last_reviewed_row value = slopx_ireland_evidence_selected_row }
		slopx_ireland_sync_cultural_evidence = yes
		slopx_ireland_sync_hidden_permissions = yes
		clr_country_flag = slopx_ireland_concealed_evidence_inspection_requested
		add_to_variable = { slopx_ireland_evidence_gui_dirty = constant:slopx_ireland.one }
	}
}
