# Ireland Total Overhaul historical and route-specific formation names.
# These groups are deliberately separate from vanilla IRE_INF_01 so a Marine
# Service or coastal formation does not consume the regular infantry numbering.

	name = "Irish Coastal Defence"
	for_countries = { IRE }
	can_use = { always = yes }
	division_types = { "infantry" }
	fallback_name = "%d Coastal Defence Battalion"
	unordered = {
		"Cork Coastal Defence Battalion"

		"Galway Coastal Defence Battalion"
		"Donegal Coastal Defence Battalion"

}
