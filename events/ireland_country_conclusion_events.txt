# Ireland family-specific postwar conclusion reports. These events are dispatched
# directly by ireland_finalize_route_conclusion and never enter the flavour pool.
add_namespace = ireland_country_conclusion

country_event = {
	id = ireland_country_conclusion.1
	title = ireland_country_conclusion.1.title
	desc = ireland_country_conclusion.1.desc
	picture = GFX_report_event_ireland_postwar_democratic_settlement
	is_triggered_only = yes
	fire_only_once = no
	trigger = { original_tag = IRE has_country_flag = ireland_family_democratic NOT = { has_country_flag = ireland_postwar_democratic_report_recorded } }
	option = {
		name = ireland_country_conclusion.1.full
		trigger = { has_country_flag = ireland_route_conclusion_full }
		custom_effect_tooltip = ireland_country_conclusion.1.full.effects
		set_country_flag = ireland_postwar_democratic_report_recorded
		set_country_flag = ireland_postwar_democratic_full_settlement
		ireland_archive_successful_route = yes
	}
	option = {
		name = ireland_country_conclusion.1.partial
		trigger = { has_country_flag = ireland_route_conclusion_partial }
		custom_effect_tooltip = ireland_country_conclusion.1.partial.effects
		set_country_flag = ireland_postwar_democratic_report_recorded
		set_country_flag = ireland_postwar_democratic_compromise_settlement
		ireland_archive_successful_route = yes
	}
	option = {
		name = ireland_country_conclusion.1.recovery
		trigger = { OR = { has_country_flag = ireland_route_conclusion_blocked has_country_flag = ireland_route_conclusion_failed } }
		custom_effect_tooltip = ireland_country_conclusion.1.recovery.effects
		set_country_flag = ireland_postwar_democratic_recovery_reported
	}
}
