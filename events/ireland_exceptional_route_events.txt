# are fired only by their focus, action, or bounded mission owners; they are
# deliberately excluded from the ordinary event dispatcher.

# `ai_event_primary = 80`, `ai_event_supported = 45`,

# `ai_event_last_resort = 2`, `ai_historical_block_factor = 0`,
# `ai_route_match_factor = 3`, and `ai_ledger_pressure_factor = 2`.
