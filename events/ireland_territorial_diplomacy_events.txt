# Ireland territorial diplomacy. Each demand stores the Irish requester, the
# controller who received it, and the state under discussion as global event
# targets because the recipient may hold the response window after the
# initiating decision chain ends. Every terminal branch clears all pointers.
add_namespace = ireland_territorial_diplomacy

@IRELAND_TERRITORIAL_WARGOAL_DAYS = 180
@IRELAND_TERRITORIAL_DIPLOMACY_AI_ACCEPT = 20
@IRELAND_TERRITORIAL_DIPLOMACY_AI_COUNTEROFFER = 35
@IRELAND_TERRITORIAL_DIPLOMACY_AI_REFUSE = 45
@IRELAND_TERRITORIAL_DIPLOMACY_STALE_REFUND = 50

# State 119: first-stage Ulster ultimatum.
country_event = {
	id = ireland_territorial_diplomacy.1
	title = ireland_territorial_diplomacy.ulster_ultimatum.t
	desc = ireland_territorial_diplomacy.ulster_ultimatum.d
	picture = GFX_report_event_ireland_territorial_ulster_ultimatum
	is_triggered_only = yes
	trigger = {
		has_event_target = ireland_territorial_diplomacy_requester
		has_event_target = ireland_territorial_diplomacy_target_state
		event_target:ireland_territorial_diplomacy_requester = {
			has_country_flag = ireland_territorial_ulster_ultimatum_pending
		}
	}
	option = {
		name = ireland_territorial_diplomacy.ulster_ultimatum.accept
		custom_effect_tooltip = ireland_territorial_diplomacy.ulster_ultimatum.accept.effects
		ai_chance = { base = @IRELAND_TERRITORIAL_DIPLOMACY_AI_ACCEPT }
		event_target:ireland_territorial_diplomacy_requester = {
			set_country_flag = ireland_territorial_ulster_ultimatum_accept_pending
			country_event = { id = ireland_territorial_diplomacy.2 }
		}
	}
	option = {
		name = ireland_territorial_diplomacy.ulster_ultimatum.counteroffer
		custom_effect_tooltip = ireland_territorial_diplomacy.ulster_ultimatum.counteroffer.effects
		ai_chance = { base = @IRELAND_TERRITORIAL_DIPLOMACY_AI_COUNTEROFFER }
		event_target:ireland_territorial_diplomacy_requester = {
			set_country_flag = ireland_territorial_ulster_ultimatum_counteroffer_pending
			country_event = { id = ireland_territorial_diplomacy.3 }
		}
	}
	option = {
		name = ireland_territorial_diplomacy.ulster_ultimatum.refuse
		custom_effect_tooltip = ireland_territorial_diplomacy.ulster_ultimatum.refuse.effects
		ai_chance = { base = @IRELAND_TERRITORIAL_DIPLOMACY_AI_REFUSE }
		event_target:ireland_territorial_diplomacy_requester = {
			set_country_flag = ireland_territorial_ulster_ultimatum_refusal_pending
			country_event = { id = ireland_territorial_diplomacy.4 }
		}
	}
}

	trigger = { original_tag = IRE has_country_flag = ireland_territorial_ulster_ultimatum_accept_pending }

		name = ireland_territorial_diplomacy.result.accept
		custom_effect_tooltip = ireland_territorial_diplomacy.ulster_ultimatum_result.accept.effects

			has_event_target = ireland_territorial_diplomacy_target_state
			event_target:ireland_territorial_diplomacy_target_state = {
				is_controlled_by = event_target:ireland_territorial_diplomacy_target_controller
